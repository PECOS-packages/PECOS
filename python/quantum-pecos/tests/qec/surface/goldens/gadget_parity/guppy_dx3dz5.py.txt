def make_memory_z(num_rounds: int):

    from guppylang.std.builtins import comptime

    @guppy
    def memory_z() -> None:
        """Z-basis memory experiment for dx=3, dz=5."""
        surf = prep_z_basis()
        init_syn = init_z_basis(surf)
        output("init_synx", init_syn)

        for _t in range(comptime(num_rounds)):
            syn = syndrome_extraction(surf)
            output("synx", syn.synx)

        final = measure_z_basis(surf)
        output("final", final)
        output("final:meas:0", final[0])
        output("final:meas:1", final[1])
